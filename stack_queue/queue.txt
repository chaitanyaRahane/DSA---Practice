#include<iostream>
#define size 3
using namespace std;
//queue (FIFO) method


class queue {
	int arr[size] = {};
	int front;
	int rear;

public:
	queue() {
		front = 0;
		rear = 0;
	}
	void enqueue(int val) {
		if (rear == size) {
			cout << "Queue is full" << endl;
		}
		else {
			arr[rear] = val;
			rear++;
		}
	}
	void dequeue() {
		if (front == rear) {
			cout << "Queue is empty" << endl;
		}
		else {
			front++;
		}
	}
	void display() {
		for (int i = front; i < rear; i++) {
			cout << arr[i] << endl;
		}
	}
};
class Node {
public:
	int data;
	Node* next;
};
class linklist {
	Node* front;
	Node* rear;

public:

	linklist() {
		front = NULL;
		rear = NULL;
	}

	void enqueue(int val) {
		Node* newNode = new Node();
		newNode->data = val;

		if (front == NULL && rear == NULL) {
			front = rear = newNode;
		}
		else {
			rear->next = newNode;
			rear = newNode;
		}
	}

	void dequeue() {
		if (front == NULL) {
			cout << "Queue is Empty" << endl;
		}
		else {
			cout << "Dequeue element" << front->data << endl;
			Node* temp = front;

			front = front->next;
			if (front == NULL) {
				rear = NULL;
			}
			delete temp;
		}
	}

};

int main(void) {
	//so 0 is front and 2 is rear in the case of linkedlist
	//0,1,2
	queue q;
	q.enqueue(0);
	q.enqueue(1);
	q.enqueue(2);
	q.dequeue();
	q.display();
	return 0;
}
