#include<iostream>

#define size 3
using namespace std;
//stack (LIFO) method
class stack {
	int arr[size] = {};
	int top;

public:
	stack() {
		top = -1;
	}

	void push(int ele) {

		if (top == size - 1) {
			cout << "stack is full" << endl;
		}
		else {
			top++;
			arr[top] = ele;
		}
	}
	void pop() {
		if (top == -1) {
			cout << "stack is empty" << endl;
		}
		else {
			cout << "Element pop" << arr[top] << endl;
			top--;
		}
	}
	void display() {
		for (int i = 0; i <= top; i++) {
			cout << arr[i] << endl;
		}
	}
};
class Node {
public:
	int data;
	Node* next;
};
class stacklistlist {
	Node* head;
	int top;
public:
	stacklistlist() {
		this->top = -1;
		this->head = nullptr;
	}
	void push(int val) {
		Node* newNodde = new Node();
		newNodde->data = val;
		newNodde->next = head;
		head = newNodde;
		top++;
	}
	void pop()
	{
		if (top == -1) {
			return;
		}
		else {
			Node* temp = head;
			head = head->next;
			delete temp;
			top--;
		}
	}
	void display(void) {
		Node* temp = head;
		while (temp != nullptr) {
			cout << temp->data << endl;
			temp = temp->next;
		}
	}
};
