#include<iostream>

using namespace std;

class Node {
public:

	Node* next;
	int data;

	Node(int value){
		this->data = value;
		this->next = nullptr;
	}
};
class LinkedList {
	Node* head;
	Node* tail;
	int length;
public:
	LinkedList(int entry) {
		Node* newNode = new Node(entry);
		head = newNode;
		tail = newNode;
		length = 1;
	}
	
	~LinkedList() {
		Node* temp = head;
		while (head != nullptr) {
			head = head->next;
			delete temp;
			temp = head;
		}
	}

	void display() {
		Node* temp = head;
		while (temp != nullptr) {
			cout << temp->data << endl;
			temp = temp->next;
		}
	}

	void append(int value) {
		Node* newNode = new Node(value);
		if (length == 0) {
			head = newNode;
			tail = newNode;
		}
		else {
			tail->next = newNode;
			tail = newNode;
			length++;
		}
	}
	void prepend(int value) {
		Node* newNode = new Node(value);
		if (length == 0) {
			head = newNode;
			tail = newNode;
		}
		else {
			newNode->next = head;
			head = newNode;
			length++;
		}
	}
	void Reverse() {
		Node* prev = nullptr;
		Node* next = nullptr;
		Node* curr = head;

		//1,2,3,4
		while (curr) {
			next = curr->next; //next = 2
			curr->next = prev; //curr->next = null;
			prev = curr; //prev = 1;
			curr = next; //curr = 2
		}
		head = prev;
	}

	void getLen() {
		cout << "Length : " << length << endl;
	}
	void deleteLast() {
		if (length == 1) {
			delete head;
			head = nullptr;
			tail = nullptr;
		}
		else {
			Node *newNode = head;
			while (newNode->next != tail)
			{
				newNode = newNode->next;
			}
			delete tail;
			tail = newNode;
			tail->next = nullptr;
		}
		length--;
	}
	void deleteFirst() {
		if (length == 1) {
			delete head;
			head = nullptr;
			tail = nullptr;
		}
		else {
			Node* temp = head;
			head = head->next;
			delete temp;
		}
		length--;
	}
	void deleteAtIndex(int index) {
		if (index == 0) {
			deleteFirst();
		}else if(index == length - 1){
			deleteLast();
		}
		else {
			Node* prev = head;

			for (int i = 0; i < index - 1; i++) {
				prev = prev->next;
			}

			Node *todelte = prev->next;
			prev->next = todelte->next;
			delete todelte;
			length--;
		}
	}
};

int main(void) {
	LinkedList* l = new LinkedList(1);
	l->append(2);
	l->append(3);
	l->append(4);

	l->prepend(0);

	l->deleteAtIndex(4);

	l->Reverse();

	l->display();

	return 0;
}
