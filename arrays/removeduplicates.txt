#include<iostream>

using namespace std;

void rmDuplicates(int arr[], int size) {

	int len = 0;
	for (int i = 0; i < size - 1; i++) {

		if (arr[i] != arr[i + 1]) {
			arr[len] = arr[i];
			len++;
		}
	}

	arr[len] = arr[size - 1];

	for (int i = 0; i < size - 1; i++) {
		cout << arr[i] << endl;
	}

}
int main(void) {

	int arr[] = { 1,1,2,3 };
	int size = sizeof(arr) / sizeof(arr[0]);

	rmDuplicates(arr, size);

	return 0;
}
