#include<iostream>

using namespace std;

void removeELE(int arr[],int size,int index) {

	for (int i = index; i < size - 1; i++) {
		arr[i] = arr[i + 1];
	}

	size--;

	for (int i = 0; i < size; i++) 
		cout << arr[i] << endl;
}
void insertEle(int arr[], int size, int pos,int ele) {

	for (int i = size; i > pos; i--) {
		arr[i] = arr[i - 1];
	}

	arr[pos] = ele;

	for (int i = 0; i < size + 1; i++)
		cout << arr[i] << endl;

}

int main(void) {
	int arr[6] = { 1,2,3,4,5 };
	//int sz = sizeof(arr) / sizeof(arr[0]);
	//removeELE(arr, sz, 2);

	insertEle(arr,5, 3, 4);

	return 0;
}
