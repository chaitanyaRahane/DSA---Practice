int hasAPairWithTarger(int arr[], int size, int target) {

	int start = 0;
	int end = size - 1;

	while (start < end) {

		int sum = arr[start] + arr[end];

		if (sum == target) {
			return 1;
		}

		if (sum < target) {
			start++;
		}
		else {
			end--;
		}
	}
	return 0;
}
int removeElement(int arr[], int size, int val) {

	int newlen = 0;

	for (int i = 0; i < size; i++) {

		if (arr[i] != val) {
			arr[newlen] = arr[i];
			newlen++;
		}
	}

	return newlen;
}

int removeDuplicates(int arr[], int size) {

	int len = 0;

	for (int i = 0; i < size - 1; i++) {
		if (arr[i] != arr[i + 1]) {
			arr[len] = arr[i];
			len++;
		}
	}
	arr[len] = arr[size - 1];
	len++;

	return len;
}
