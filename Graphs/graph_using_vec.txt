#include<iostream>
#include<vector>
#include<queue>

using namespace std;

class Graph {
	int V;
	vector<vector<int>> adj;

public:
	Graph(int V) {
		this->V = V;
		adj.resize(V);
	}

	void addEdges(int u,int v) {
		adj[u].push_back(v);
		adj[v].push_back(u);
	}

	void printList() {
		for (int i = 0; i < V; i++) {
			cout << "Vertex : " << i << ":";

			for (int neighbor : adj[i]) {
				cout << " ->" << neighbor;
			}
			cout << endl;
		}
	}

	void BFS(int start) {
		vector<bool> visited(V, false);
		queue<int> q;
			
		visited[start] = true;
		q.push(start);

		cout << "BFS starting from vertex" <<start<< ":";

		while (!q.empty()) {
			int node = q.front();
			q.pop();

			cout << node << endl;

			for (int neighbor : adj[node]) {
				if (!visited[neighbor]) {
					visited[neighbor] = true;
					q.push(neighbor);
				}
			}

		}
	}
	void DFSUtils(int node,vector<bool>& visited) {
		visited[node] = true;
		cout << node << " ";

		for (int neighbor : adj[node]) {
			if (!visited[neighbor]) {
				DFSUtils(neighbor, visited);
			}
		}

	}
	void DFS(int u) {
		vector<bool> visited(V, false);
		cout << "DFS starting from vertex" << u << ":";
		DFSUtils(u, visited);
		cout << endl;
	}

};
int main(void) {

	Graph g(5);

	g.addEdges(0, 1);
	g.addEdges(0, 2);
	
	g.addEdges(1, 3);
	g.addEdges(1, 4);

	g.printList();
	g.BFS(0);
	g.DFS(0);

	return 0;
}
