#include<iostream>

using namespace std;

void linearSearch(int arr[], int size, int key) {

	for (int i = 0; i < size; i++) {
		if (arr[i] == key) {
			cout << "found it!" << endl;
			return;
		}
	}

}
//binary search ->divide the data into two parts using start and end
//but it should be a sorted array
void binarySearch(int arr[], int start, int end, int key) {

	while (start <= end) {

		int mid = (start + end) / 2;

		if (arr[mid] == key)
		{
			cout << "Found it!" << endl;
			break;
		}

		if (arr[mid] < key)
		{
			start = mid + 1;
		}
		else {
			end = mid - 1;
		}
	}
}
