#include<iostream>

using namespace std;

void bubblesort(int arr[],int size) {
	bool swapped;

	for (int i = 0; i < size - 1; i++) {
		swapped = false;
		for (int j = 0; j < size - i - 1; j++) {
			if (arr[j] > arr[j + 1]) {
				swap(arr[j],arr[j+1]);
				swapped = true;
			}
			
		}
		if (!swapped) break; // if already swapped
	}

	for (int i = 0; i < size; i++) {
		cout << arr[i] << endl;
	}

	//1st iteration of i -> i = 0 , size = 2
	//1st iteration of j -> j = 0 , size = 2
	// 9 and 8 will get swapped
	// result = 8 , 9 , 7
	
	//2nd iteration of j -> j = 1, size = 2
	//now 9 and 7 will get swapped.
	//result is 8 , 7 , 9, and j++ which is j = 2 breaks the inner loop

	//2nd iteration of i -> i = 1
	//3rd iteration of j -> j = 0 , size = 1
	// so 8 and 7 will be swapped 
	//result 7 , 8 , 9

}
int main(void) {
	int arr[] = { 9,8,7};
	int size = sizeof(arr) / sizeof(arr[0]);

	bubblesort(arr, size);

	return 0;
}
