void ourswap(int arr1[], int arr2[]);
void bubblesort(int arr[],int size) {

	for (int i = 0; i < size - 1; i++) {

		for (int j = 0; j < size - 1; j++) {
			if (arr[j] > arr[j + 1])
				ourswap(&arr[j], &arr[j + 1]);
		}
	}

	for (int i = 0; i < size; i++) {
		cout << arr[i] << endl;
	}
}
void ourswap(int arr1[], int arr2[]) {

	int temp = *arr1;
	*arr1 = *arr2;
	*arr2 = temp;
}
int main(void) {

	int arr[] = { 50,25,5,20,10 };
	int size = sizeof(arr) / sizeof(arr[0]);

	bubblesort(arr, size);

	return 0;
}
