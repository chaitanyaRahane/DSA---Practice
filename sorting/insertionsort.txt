#include<iostream>

using namespace std;

void insertionsort(int arr[], int size) {

	for (int i = 1; i < size; i++) {
		int key = arr[i];
		int j = i - 1;

		while (j >= 0 && arr[j] > key) {
			arr[j + 1] = arr[j];
			j--;
		}
		arr[j + 1] = key;

	}

	for (int i = 0; i < size; i++) {
		cout << arr[i] << endl;
	}
}
int main(void) {
	int arr[] = {5,4,3};
	int size = sizeof(arr) / sizeof(arr[0]);

	insertionsort(arr, size);

	return 0;
}
